cmake_minimum_required(VERSION 3.20)
project(fortran_test_providers LANGUAGES C)
# Controlled local test doubles, never production providers or performance data.
get_filename_component(_tests "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/include")
foreach(_name fftw3.f03 mkl_spblas.f90 mkl_service.h)
  configure_file("${CMAKE_CURRENT_SOURCE_DIR}/${_name}" "${CMAKE_BINARY_DIR}/include/${_name}" COPYONLY)
endforeach()
configure_file("${_tests}/fixtures/fftw/include/fftw3.h" "${CMAKE_BINARY_DIR}/include/fftw3.h" COPYONLY)
add_library(mkl_rt STATIC "${_tests}/fixtures/cpu_libs/fake_cpu_providers.c" fortran_abi.c)
target_include_directories(mkl_rt PRIVATE "${_tests}/fixtures/cpu_libs/include")
add_library(fftw3f STATIC "${_tests}/fixtures/fftw/fake_fftw3f.c" fftw_abi.c)
target_include_directories(fftw3f PRIVATE "${_tests}/fixtures/fftw/include")
add_library(fftw3f_threads STATIC "${_tests}/fixtures/fftw/fake_fftw3f_threads.c")
target_include_directories(fftw3f_threads PRIVATE "${_tests}/fixtures/fftw/include")
