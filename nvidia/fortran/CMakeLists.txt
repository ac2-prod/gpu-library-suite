# Fortran-only build tree: target stems intentionally match teaching filenames.
set(_common "${CMAKE_SOURCE_DIR}/common/fortran")
set(_fortran_thrust_include "")
foreach(_cuda_include IN LISTS CUDAToolkit_INCLUDE_DIRS)
  if(EXISTS "${_cuda_include}/cccl/thrust/version.h")
    set(_fortran_thrust_include "${_cuda_include}/cccl")
    break()
  elseif(EXISTS "${_cuda_include}/thrust/version.h")
    set(_fortran_thrust_include "${_cuda_include}")
  endif()
endforeach()
add_library(gpu_suite_fortran_support STATIC "${_common}/benchmark_support.f90")
set_target_properties(gpu_suite_fortran_support PROPERTIES
  Fortran_MODULE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/modules/support")
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/modules/support")
target_include_directories(gpu_suite_fortran_support PUBLIC "${CMAKE_CURRENT_BINARY_DIR}/modules/support")
target_compile_options(gpu_suite_fortran_support PRIVATE ${GPU_SUITE_FORTRAN_SAFE_OPTIONS})
add_library(gpu_suite_fortran_matrices STATIC "${_common}/make_dense_system.f90" "${_common}/make_poisson2d_csr.f90")
target_compile_options(gpu_suite_fortran_matrices PRIVATE ${GPU_SUITE_FORTRAN_SAFE_OPTIONS})

if(GPU_SUITE_BUILD_CPU)
  find_package(FFTW3f QUIET)
  find_path(GPU_SUITE_FORTRAN_FFTW_INCLUDE NAMES fftw3.f03
    HINTS "${FFTW3f_INCLUDE_DIR}" "${FFTW_ROOT}" ENV FFTW_ROOT PATH_SUFFIXES include)
  find_library(GPU_SUITE_FORTRAN_MKL_LIBRARY NAMES mkl_rt
    HINTS ENV MKLROOT PATH_SUFFIXES lib lib64)
  find_path(GPU_SUITE_FORTRAN_MKL_INCLUDE NAMES mkl_spblas.f90
    HINTS ENV MKLROOT PATH_SUFFIXES include)
  find_path(GPU_SUITE_FORTRAN_MKL_SERVICE_INCLUDE NAMES mkl_service.h
    HINTS ENV MKLROOT PATH_SUFFIXES include)
  find_package(Threads QUIET)
  if(GPU_SUITE_FORTRAN_MKL_LIBRARY AND GPU_SUITE_FORTRAN_MKL_SERVICE_INCLUDE)
    include(GpuSuiteProbe)
    set(_mkl_service_links "${GPU_SUITE_FORTRAN_MKL_LIBRARY}" ${CMAKE_DL_LIBS})
    if(TARGET Threads::Threads)
      list(APPEND _mkl_service_links Threads::Threads)
    endif()
    gpu_suite_check_c_link(GPU_SUITE_FORTRAN_MKL_SERVICE_LINKS
      SOURCE "#include <mkl_service.h>\nint main(void) { char version[256]; mkl_set_num_threads(1); mkl_get_version_string(version,256); return 0; }"
      INCLUDES "${GPU_SUITE_FORTRAN_MKL_SERVICE_INCLUDE}"
      LIBRARIES ${_mkl_service_links})
  endif()
endif()

set(_libraries cufft cublas cusparse cusolver curand thrust)
set(_stems fft blas sparse solver rand reduce)
foreach(_index RANGE 0 5)
  list(GET _libraries ${_index} _library)
  list(GET _stems ${_index} _stem)
  set(_helper "")
  if(_library STREQUAL "cusparse")
    set(_helper "${_common}/make_poisson2d_csr.f90")
  elseif(_library STREQUAL "cusolver")
    set(_helper "${_common}/make_dense_system.f90")
  endif()
  foreach(_impl RANGE 0 2)
    set(_enabled FALSE)
    set(_reason "not requested")
    set(_options "")
    set(_links "")
    set(_includes "")
    set(_definitions "GPU_SUITE_IMPL=${_impl}")
    set(_backend "")
    set(_backends "")
    set(_wrapper "")
    if(_impl EQUAL 0)
      set(_implementation cpu)
      set(_name "${_stem}_cpu")
      set(_profile cpu-cuda)
      if(GPU_SUITE_BUILD_CPU)
        set(_enabled TRUE)
        set(_reason "")
        if(_index EQUAL 0)
          set(_backend cpu-fftw-serial)
          if(FFTW3f_FOUND AND GPU_SUITE_FORTRAN_FFTW_INCLUDE)
            set(_links FFTW3f::fftw3f)
            set(_includes "${GPU_SUITE_FORTRAN_FFTW_INCLUDE}")
            list(APPEND _definitions GPU_SUITE_FORTRAN_FFTW=1)
          else()
            set(_enabled FALSE)
            set(_reason "FFTW3f library and fftw3.f03 are required")
          endif()
        elseif(_index LESS 4)
          set(_backend cpu-onemkl)
          if(GPU_SUITE_FORTRAN_MKL_LIBRARY AND GPU_SUITE_FORTRAN_MKL_SERVICE_LINKS)
            set(_links "${GPU_SUITE_FORTRAN_MKL_LIBRARY}" ${CMAKE_DL_LIBS})
            set(_includes "${GPU_SUITE_FORTRAN_MKL_SERVICE_INCLUDE}")
            if(TARGET Threads::Threads)
              list(APPEND _links Threads::Threads)
            endif()
            list(APPEND _definitions GPU_SUITE_FORTRAN_MKL=1)
            if(_index EQUAL 2)
              if(GPU_SUITE_FORTRAN_MKL_INCLUDE)
                list(APPEND _includes "${GPU_SUITE_FORTRAN_MKL_INCLUDE}")
              else()
                set(_enabled FALSE)
                set(_reason "oneMKL mkl_spblas.f90 is required (not another compiler's .mod)")
              endif()
            endif()
          else()
            set(_enabled FALSE)
            set(_reason "oneMKL LP64 mkl_rt and mkl_service.h/version/thread symbols are required")
          endif()
        elseif(_index EQUAL 4)
          set(_backend cpu-fortran-random-serial)
        else()
          set(_backend cpu-fortran-sum-serial)
        endif()
        set(_backends "${_backend}")
      endif()
    else()
      set(_links CUDA::cudart)
      if(_index LESS 5)
        list(APPEND _links "CUDA::${_library}")
      endif()
      set(_options ${GPU_SUITE_FORTRAN_GPU_OPTIONS})
      if(_index LESS 5)
        list(APPEND _options "-cudalib=${_library}")
      else()
        list(APPEND _options -c++libs)
      endif()
      if(_impl EQUAL 1)
        set(_name "${_stem}_gpu")
        set(_implementation cuda)
        set(_profile cpu-cuda)
        set(_enabled "${GPU_SUITE_CUDA_TARGETS_ENABLED}")
        set(_reason "${GPU_SUITE_CUDA_DISABLE_REASON}")
      else()
        set(_name "openacc_${_library}")
        set(_implementation openacc)
        set(_profile openacc)
        set(_enabled "${GPU_SUITE_OPENACC_TARGETS_ENABLED}")
        set(_reason "${GPU_SUITE_OPENACC_DISABLE_REASON}")
        list(APPEND _options -acc=gpu)
      endif()
      if(_index EQUAL 5 AND _enabled)
        if(GPU_SUITE_CUDA_LANGUAGE_ENABLED AND _fortran_thrust_include)
          set(_wrapper "${CMAKE_CURRENT_SOURCE_DIR}/thrust/examples/thrust_wrapper.cu")
          list(APPEND _includes "${_fortran_thrust_include}")
        else()
          set(_enabled FALSE)
          set(_reason "Thrust interop requires NVCC, explicit CMAKE_CUDA_ARCHITECTURES and external Toolkit Thrust/CCCL headers")
        endif()
      endif()
    endif()
    set(_source "${CMAKE_CURRENT_SOURCE_DIR}/${_library}/examples/${_name}.f90")
    if(_enabled AND NOT _wrapper)
      gpu_suite_fortran_probe(_probe_${_name} SOURCES "${_source}" ${_helper}
        LIBRARIES ${_links} OPTIONS ${_options} INCLUDES ${_includes})
      if(NOT _probe_${_name})
        set(_enabled FALSE)
        set(_reason "Fortran source/module/symbol compile-and-link probe failed (fortran-probes/_probe_${_name}.log)")
      endif()
    endif()
    if(_enabled AND _impl EQUAL 0 AND _index EQUAL 3)
      # The teaching example uses dgesv; benchmarks require both staged APIs.
      set(_solver_probe "${CMAKE_BINARY_DIR}/fortran-probes/solver_staged.f90")
      file(WRITE "${_solver_probe}" "program probe\n real(8) :: a(1,1), b(1,1)\n integer :: piv(1), info\n a=1; b=1\n call dgetrf(1,1,a,1,piv,info)\n call dgetrs('N',1,1,a,1,piv,b,1,info)\nend program\n")
      gpu_suite_fortran_probe(_probe_solver_staged SOURCES "${_solver_probe}" LIBRARIES ${_links})
      if(NOT _probe_solver_staged)
        set(_enabled FALSE)
        set(_reason "oneMKL Fortran dgetrf/dgetrs compile-and-link probe failed")
      endif()
    endif()
    # Thrust has a separate CUDA C++ compile/link probe; the complete mixed
    # language link is also exercised by the ordinary target build.
    if(_enabled AND _wrapper)
      include(GpuSuiteProbe)
      file(READ "${_wrapper}" _wrapper_source)
      string(APPEND _wrapper_source "\nint main() { return thrust_square_sum(nullptr,0)==0 ? 0 : 1; }\n")
      gpu_suite_check_cuda_link(_fortran_thrust_links SOURCE "${_wrapper_source}"
        INCLUDES "${_fortran_thrust_include}" LIBRARIES CUDA::cudart)
      if(NOT _fortran_thrust_links)
        set(_enabled FALSE)
        set(_reason "Thrust CUDA C++ wrapper did not compile/link")
      endif()
    endif()
    if(_enabled)
      set(_variant "fortran-${_implementation}-${_library}")
      if(_impl EQUAL 0)
        set(_variant "fortran-${_backend}")
      endif()
      foreach(_role example benchmark)
        if(_role STREQUAL "example")
          if(NOT GPU_SUITE_BUILD_EXAMPLES)
            continue()
          endif()
          set(_target "${_name}")
          set(_target_source "${_source}")
          set(_sources "${_source}" ${_helper} ${_wrapper})
        else()
          if(NOT GPU_SUITE_BUILD_BENCHMARKS)
            continue()
          endif()
          set(_target "${_name}_bench")
          set(_target_source "${CMAKE_CURRENT_SOURCE_DIR}/${_library}/benchmarks/${_target}.f90")
          set(_sources "${_target_source}" "${CMAKE_CURRENT_SOURCE_DIR}/${_library}/benchmarks/${_stem}_workload.F90"
            "${_common}/benchmark_bridge.cpp" ${_wrapper})
        endif()
        add_executable(${_target} ${_sources})
        set(_modules "${CMAKE_CURRENT_BINARY_DIR}/modules/${_target}")
        file(MAKE_DIRECTORY "${_modules}")
        set_target_properties(${_target} PROPERTIES Fortran_MODULE_DIRECTORY "${_modules}"
          LINKER_LANGUAGE Fortran Fortran_PREPROCESS ON
          RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/${_library}")
        target_include_directories(${_target} BEFORE PRIVATE "${_modules}" ${_includes})
        target_compile_definitions(${_target} PRIVATE ${_definitions}
          "GPU_SUITE_FORTRAN_CPU_BACKEND=\"${_backend}\"")
        foreach(_option IN LISTS GPU_SUITE_FORTRAN_SAFE_OPTIONS _options)
          target_compile_options(${_target} PRIVATE "$<$<COMPILE_LANGUAGE:Fortran>:${_option}>")
        endforeach()
        target_link_options(${_target} PRIVATE ${GPU_SUITE_FORTRAN_SAFE_OPTIONS} ${_options})
        target_link_libraries(${_target} PRIVATE ${_links})
        set(_target_backends ${_backends})
        if(_role STREQUAL "benchmark")
          target_link_libraries(${_target} PRIVATE gpu_suite_fortran_support gpu_suite_fortran_matrices gpu_suite::common)
          if(_impl EQUAL 0 AND _index EQUAL 0 AND FFTW3f_THREADS_FOUND)
            target_link_libraries(${_target} PRIVATE FFTW3f::threads)
            target_compile_definitions(${_target} PRIVATE GPU_SUITE_HAVE_FFTW_THREADS=1)
            list(APPEND _target_backends cpu-fftw-threaded)
          endif()
        endif()
        gpu_suite_configure_executable(${_target} "${_target_source}"
          LIBRARY "${_library}" IMPLEMENTATION "${_implementation}" ROLE "${_role}"
          BUILD_PROFILE "${_profile}" BACKEND_VARIANT "${_variant}" CPU_BACKENDS ${_target_backends})
      endforeach()
    endif()
    gpu_suite_record_status("Fortran ${_library} ${_implementation}" "${_enabled}" "${_reason}")
  endforeach()
endforeach()
