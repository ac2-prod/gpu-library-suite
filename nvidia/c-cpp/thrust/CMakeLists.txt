include(GpuSuiteProbe)

set(_thrust_cccl_include_dir "")
foreach(_cuda_include_dir IN LISTS CUDAToolkit_INCLUDE_DIRS)
  if(EXISTS "${_cuda_include_dir}/cccl/thrust/version.h")
    set(_thrust_cccl_include_dir "${_cuda_include_dir}/cccl")
    break()
  endif()
endforeach()

function(_gpu_suite_thrust_use_external_cccl target)
  target_include_directories(${target} BEFORE PRIVATE
    "${_thrust_cccl_include_dir}")
endfunction()

set(_cpu_enabled "${GPU_SUITE_BUILD_CPU}")
set(_cpu_variant stl-serial)
set(_cpu_backend cpu-stl-serial)
set(_cpu_reason "GPU_SUITE_BUILD_CPU is OFF")
if(GPU_SUITE_BUILD_CPU AND GPU_SUITE_CPU_THRUST_BACKEND STREQUAL "OPENMP")
  find_package(OpenMP COMPONENTS CXX QUIET)
  if(OpenMP_CXX_FOUND)
    set(_cpu_variant openmp)
    set(_cpu_backend cpu-openmp)
    set(_cpu_reason "OpenMP CXX compile-and-link support")
  else()
    set(_cpu_enabled FALSE)
    set(_cpu_reason "OpenMP CXX compile-and-link support was not found")
  endif()
elseif(GPU_SUITE_BUILD_CPU AND
       NOT GPU_SUITE_CPU_THRUST_BACKEND STREQUAL "STL")
  set(_cpu_enabled FALSE)
  set(_cpu_reason "invalid GPU_SUITE_CPU_THRUST_BACKEND")
elseif(GPU_SUITE_BUILD_CPU)
  set(_cpu_reason "dependency-free stl-serial")
endif()
if(_cpu_enabled AND GPU_SUITE_BUILD_EXAMPLES)
  add_executable(reduce_cpu examples/reduce_cpu.cpp)
  if(_cpu_variant STREQUAL "openmp")
    target_link_libraries(reduce_cpu PRIVATE OpenMP::OpenMP_CXX)
    target_compile_definitions(reduce_cpu PRIVATE GPU_SUITE_USE_OPENMP=1)
  endif()
  gpu_suite_configure_executable(reduce_cpu examples/reduce_cpu.cpp LIBRARY thrust IMPLEMENTATION cpu ROLE example BUILD_PROFILE cpu-cuda BACKEND_VARIANT "${_cpu_variant}" CPU_BACKENDS "${_cpu_backend}")
endif()
if(_cpu_enabled AND GPU_SUITE_BUILD_BENCHMARKS)
  add_executable(reduce_cpu_bench benchmarks/reduce_cpu_bench.cpp)
  target_link_libraries(reduce_cpu_bench PRIVATE gpu_suite::common)
  if(_cpu_variant STREQUAL "openmp")
    target_link_libraries(reduce_cpu_bench PRIVATE OpenMP::OpenMP_CXX)
    target_compile_definitions(reduce_cpu_bench PRIVATE GPU_SUITE_USE_OPENMP=1)
  endif()
  target_compile_definitions(reduce_cpu_bench PRIVATE "GPU_SUITE_COMPILED_CPU_BACKEND=\"${_cpu_backend}\"")
  gpu_suite_configure_executable(reduce_cpu_bench benchmarks/reduce_cpu_bench.cpp LIBRARY thrust IMPLEMENTATION cpu ROLE benchmark BUILD_PROFILE cpu-cuda BACKEND_VARIANT "${_cpu_variant}" CPU_BACKENDS "${_cpu_backend}")
endif()
set(_probe [=[
#include <cuda_runtime_api.h>
#include <thrust/device_ptr.h>
#include <thrust/transform_reduce.h>
#include <thrust/functional.h>
struct square { __host__ __device__ double operator()(double x) const { return x*x; } };
int main() { double *p=0; thrust::device_ptr<double> d=thrust::device_pointer_cast(p); (void)thrust::transform_reduce(d,d,square{},0.0,thrust::plus<double>()); return cudaFree(0)==cudaSuccess ? 0 : 1; }
]=])
set(_cuda_enabled FALSE)
set(_cuda_reason "direct CUDA targets are disabled")
if(GPU_SUITE_CUDA_TARGETS_ENABLED AND TARGET CUDA::cudart)
  if("${_thrust_cccl_include_dir}" STREQUAL "")
    set(_cuda_reason
        "external CUDAToolkit include/cccl/thrust/version.h was not found")
  else()
    gpu_suite_check_cuda_link(GPU_SUITE_THRUST_CUDA_LINKS
      SOURCE "${_probe}"
      INCLUDES "${_thrust_cccl_include_dir}"
      LIBRARIES CUDA::cudart)
    if(GPU_SUITE_THRUST_CUDA_LINKS)
      set(_cuda_enabled TRUE)
      set(_cuda_reason "")
    else()
      set(_cuda_reason "CUDA Thrust transform_reduce did not compile and link")
    endif()
  endif()
endif()
if(_cuda_enabled AND GPU_SUITE_BUILD_EXAMPLES)
  add_executable(reduce_gpu examples/reduce_gpu.cu)
  _gpu_suite_thrust_use_external_cccl(reduce_gpu)
  target_link_libraries(reduce_gpu PRIVATE CUDA::cudart)
  gpu_suite_configure_executable(reduce_gpu examples/reduce_gpu.cu LIBRARY thrust IMPLEMENTATION cuda ROLE example BUILD_PROFILE cpu-cuda BACKEND_VARIANT cuda-thrust)
endif()
if(_cuda_enabled AND GPU_SUITE_BUILD_BENCHMARKS)
  add_executable(reduce_gpu_bench benchmarks/reduce_gpu_bench.cu)
  _gpu_suite_thrust_use_external_cccl(reduce_gpu_bench)
  target_link_libraries(reduce_gpu_bench PRIVATE gpu_suite::common CUDA::cudart)
  gpu_suite_configure_executable(reduce_gpu_bench benchmarks/reduce_gpu_bench.cu LIBRARY thrust IMPLEMENTATION cuda ROLE benchmark BUILD_PROFILE cpu-cuda BACKEND_VARIANT cuda-thrust)
endif()
set(_openacc_enabled FALSE)
set(_openacc_reason "OpenACC targets are disabled")
if(GPU_SUITE_OPENACC_TARGETS_ENABLED AND TARGET OpenACC::OpenACC_CXX AND TARGET CUDA::cudart)
  if("${_thrust_cccl_include_dir}" STREQUAL "")
    set(_openacc_reason
        "external CUDAToolkit include/cccl/thrust/version.h was not found")
  else()
    gpu_suite_check_cxx_link(GPU_SUITE_THRUST_OPENACC_LINKS
      SOURCE "${_probe}"
      INCLUDES "${_thrust_cccl_include_dir}"
      COMPILE_OPTIONS -cuda
      LINK_OPTIONS -cuda
      LIBRARIES OpenACC::OpenACC_CXX CUDA::cudart)
    if(GPU_SUITE_THRUST_OPENACC_LINKS)
      set(_openacc_enabled TRUE)
      set(_openacc_reason "")
    else()
      set(_openacc_reason "nvc++ -cuda OpenACC Thrust probe failed")
    endif()
  endif()
endif()
if(_openacc_enabled AND GPU_SUITE_BUILD_EXAMPLES)
  add_executable(openacc_thrust examples/openacc_thrust.cpp)
  _gpu_suite_thrust_use_external_cccl(openacc_thrust)
  target_compile_options(openacc_thrust PRIVATE -cuda)
  target_link_options(openacc_thrust PRIVATE -cuda)
  target_link_libraries(openacc_thrust PRIVATE OpenACC::OpenACC_CXX CUDA::cudart)
  gpu_suite_configure_executable(openacc_thrust examples/openacc_thrust.cpp LIBRARY thrust IMPLEMENTATION openacc ROLE example BUILD_PROFILE openacc BACKEND_VARIANT nvhpc-cuda-thrust)
endif()
if(_openacc_enabled AND GPU_SUITE_BUILD_BENCHMARKS)
  add_executable(openacc_thrust_bench benchmarks/openacc_thrust_bench.cpp)
  _gpu_suite_thrust_use_external_cccl(openacc_thrust_bench)
  target_compile_options(openacc_thrust_bench PRIVATE -cuda)
  target_link_options(openacc_thrust_bench PRIVATE -cuda)
  target_link_libraries(openacc_thrust_bench PRIVATE gpu_suite::common OpenACC::OpenACC_CXX CUDA::cudart)
  gpu_suite_configure_executable(openacc_thrust_bench benchmarks/openacc_thrust_bench.cpp LIBRARY thrust IMPLEMENTATION openacc ROLE benchmark BUILD_PROFILE openacc BACKEND_VARIANT nvhpc-cuda-thrust)
endif()
gpu_suite_record_status("Thrust CPU targets" "${_cpu_enabled}" "${_cpu_reason}")
gpu_suite_record_status("Thrust CUDA targets" "${_cuda_enabled}" "${_cuda_reason}")
gpu_suite_record_status("Thrust OpenACC targets" "${_openacc_enabled}" "${_openacc_reason}")
