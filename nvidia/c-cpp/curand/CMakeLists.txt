include(GpuSuiteProbe)

set(_cpu_enabled "${GPU_SUITE_BUILD_CPU}")
set(_cpu_reason "GPU_SUITE_BUILD_CPU is OFF")
if(GPU_SUITE_BUILD_CPU)
  set(_cpu_reason "dependency-free C++17")
endif()
if(GPU_SUITE_BUILD_CPU AND GPU_SUITE_BUILD_EXAMPLES)
  add_executable(rand_cpu examples/rand_cpu.cpp)
  gpu_suite_configure_executable(rand_cpu examples/rand_cpu.cpp LIBRARY curand IMPLEMENTATION cpu ROLE example BUILD_PROFILE cpu-cuda BACKEND_VARIANT std-mt19937-64 CPU_BACKENDS cpu-std-random-serial)
endif()
if(GPU_SUITE_BUILD_CPU AND GPU_SUITE_BUILD_BENCHMARKS)
  add_executable(rand_cpu_bench benchmarks/rand_cpu_bench.cpp)
  target_link_libraries(rand_cpu_bench PRIVATE gpu_suite::common)
  gpu_suite_configure_executable(rand_cpu_bench benchmarks/rand_cpu_bench.cpp LIBRARY curand IMPLEMENTATION cpu ROLE benchmark BUILD_PROFILE cpu-cuda BACKEND_VARIANT std-mt19937-64 CPU_BACKENDS cpu-std-random-serial)
endif()
set(_probe [=[
#include <cuda_runtime_api.h>
#include <curand.h>
int main() {
  curandGenerator_t generator;
  if (cudaFree(0) != cudaSuccess) return 1;
  if (curandCreateGenerator(&generator,CURAND_RNG_PSEUDO_DEFAULT) != CURAND_STATUS_SUCCESS) return 2;
  (void)curandSetPseudoRandomGeneratorSeed(generator,1234ULL);
  (void)curandSetGeneratorOffset(generator,0ULL);
  (void)curandGenerateUniformDouble(generator,0,1);
  return curandDestroyGenerator(generator) == CURAND_STATUS_SUCCESS ? 0 : 3;
}
]=])
set(_cuda_enabled FALSE)
set(_cuda_reason "direct CUDA targets are disabled")
if(GPU_SUITE_CUDA_TARGETS_ENABLED AND TARGET CUDA::cudart AND TARGET CUDA::curand)
  gpu_suite_check_cuda_link(GPU_SUITE_CURAND_CUDA_LINKS SOURCE "${_probe}" LIBRARIES CUDA::cudart CUDA::curand)
  if(GPU_SUITE_CURAND_CUDA_LINKS)
    set(_cuda_enabled TRUE)
    set(_cuda_reason "")
  else()
    set(_cuda_reason "CUDA Runtime/cuRAND generator symbols did not link")
  endif()
endif()
if(_cuda_enabled AND GPU_SUITE_BUILD_EXAMPLES)
  add_executable(rand_gpu examples/rand_gpu.cu)
  target_link_libraries(rand_gpu PRIVATE CUDA::cudart CUDA::curand)
  gpu_suite_configure_executable(rand_gpu examples/rand_gpu.cu LIBRARY curand IMPLEMENTATION cuda ROLE example BUILD_PROFILE cpu-cuda BACKEND_VARIANT cuda-curand-default)
endif()
if(_cuda_enabled AND GPU_SUITE_BUILD_BENCHMARKS)
  add_executable(rand_gpu_bench benchmarks/rand_gpu_bench.cu)
  target_link_libraries(rand_gpu_bench PRIVATE gpu_suite::common CUDA::cudart CUDA::curand)
  gpu_suite_configure_executable(rand_gpu_bench benchmarks/rand_gpu_bench.cu LIBRARY curand IMPLEMENTATION cuda ROLE benchmark BUILD_PROFILE cpu-cuda BACKEND_VARIANT cuda-curand-default)
endif()
set(_openacc_enabled FALSE)
set(_openacc_reason "OpenACC targets are disabled")
if(GPU_SUITE_OPENACC_TARGETS_ENABLED AND TARGET OpenACC::OpenACC_CXX AND TARGET CUDA::cudart AND TARGET CUDA::curand)
  gpu_suite_check_cxx_link(GPU_SUITE_CURAND_OPENACC_LINKS SOURCE "${_probe}" LIBRARIES OpenACC::OpenACC_CXX CUDA::cudart CUDA::curand)
  if(GPU_SUITE_CURAND_OPENACC_LINKS)
    set(_openacc_enabled TRUE)
    set(_openacc_reason "")
  else()
    set(_openacc_reason "nvc++ OpenACC CUDA Runtime/cuRAND probe failed")
  endif()
endif()
if(_openacc_enabled AND GPU_SUITE_BUILD_EXAMPLES)
  add_executable(openacc_curand examples/openacc_curand.cpp)
  target_link_libraries(openacc_curand PRIVATE OpenACC::OpenACC_CXX CUDA::cudart CUDA::curand)
  gpu_suite_configure_executable(openacc_curand examples/openacc_curand.cpp LIBRARY curand IMPLEMENTATION openacc ROLE example BUILD_PROFILE openacc BACKEND_VARIANT nvhpc-curand-default)
endif()
if(_openacc_enabled AND GPU_SUITE_BUILD_BENCHMARKS)
  add_executable(openacc_curand_bench benchmarks/openacc_curand_bench.cpp)
  target_link_libraries(openacc_curand_bench PRIVATE gpu_suite::common OpenACC::OpenACC_CXX CUDA::cudart CUDA::curand)
  gpu_suite_configure_executable(openacc_curand_bench benchmarks/openacc_curand_bench.cpp LIBRARY curand IMPLEMENTATION openacc ROLE benchmark BUILD_PROFILE openacc BACKEND_VARIANT nvhpc-curand-default)
endif()
gpu_suite_record_status("cuRAND CPU targets" "${_cpu_enabled}" "${_cpu_reason}")
gpu_suite_record_status("cuRAND CUDA targets" "${_cuda_enabled}" "${_cuda_reason}")
gpu_suite_record_status("cuRAND OpenACC targets" "${_openacc_enabled}" "${_openacc_reason}")
