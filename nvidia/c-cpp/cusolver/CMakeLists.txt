include(GpuSuiteCpuDependencies)
include(GpuSuiteProbe)

set(_cpu_enabled FALSE)
set(_cpu_reason "GPU_SUITE_BUILD_CPU is OFF")
if(GPU_SUITE_BUILD_CPU)
  gpu_suite_find_lapacke()
  if(GPU_SUITE_LAPACKE_FOUND)
    set(_cpu_enabled TRUE)
    set(_cpu_reason "")
    string(TOLOWER "${GPU_SUITE_LAPACKE_PROVIDER}" _cpu_variant)
    if(_cpu_variant STREQUAL "generic_lapacke")
      set(_cpu_backend cpu-generic-lapacke)
    elseif(_cpu_variant STREQUAL "openblas")
      set(_cpu_backend cpu-openblas)
    else()
      set(_cpu_backend cpu-onemkl)
    endif()
  else()
    set(_cpu_reason "${GPU_SUITE_LAPACKE_REASON}")
  endif()
endif()
if(_cpu_enabled AND GPU_SUITE_BUILD_EXAMPLES)
  add_executable(solver_cpu examples/solver_cpu.c)
  target_link_libraries(solver_cpu PRIVATE GpuSuite::LAPACKE)
  gpu_suite_configure_executable(solver_cpu examples/solver_cpu.c LIBRARY cusolver IMPLEMENTATION cpu ROLE example BUILD_PROFILE cpu-cuda BACKEND_VARIANT "${_cpu_variant}" CPU_BACKENDS "${_cpu_backend}")
endif()
if(_cpu_enabled AND GPU_SUITE_BUILD_BENCHMARKS)
  add_executable(solver_cpu_bench benchmarks/solver_cpu_bench.c)
  target_link_libraries(solver_cpu_bench PRIVATE gpu_suite::common GpuSuite::LAPACKE)
  target_compile_definitions(solver_cpu_bench PRIVATE "GPU_SUITE_COMPILED_CPU_BACKEND=\"${_cpu_backend}\"")
  gpu_suite_configure_executable(solver_cpu_bench benchmarks/solver_cpu_bench.c LIBRARY cusolver IMPLEMENTATION cpu ROLE benchmark BUILD_PROFILE cpu-cuda BACKEND_VARIANT "${_cpu_variant}" CPU_BACKENDS "${_cpu_backend}")
endif()

set(_probe [=[
#include <cuda_runtime_api.h>
#include <cusolverDn.h>
int main() {
  cusolverDnHandle_t handle;
  int workspace=0;
  if (cudaFree(0) != cudaSuccess) return 1;
  if (cusolverDnCreate(&handle) != CUSOLVER_STATUS_SUCCESS) return 2;
  (void)cusolverDnDgetrf_bufferSize(handle,1,1,0,1,&workspace);
  (void)cusolverDnDgetrf(handle,1,1,0,1,0,0,0);
  (void)cusolverDnDgetrs(handle,CUBLAS_OP_N,1,1,0,1,0,0,1,0);
  return cusolverDnDestroy(handle) == CUSOLVER_STATUS_SUCCESS ? 0 : 3;
}
]=])
set(_cuda_enabled FALSE)
set(_cuda_reason "direct CUDA targets are disabled")
if(GPU_SUITE_CUDA_TARGETS_ENABLED AND TARGET CUDA::cudart AND TARGET CUDA::cusolver)
  gpu_suite_check_cuda_link(GPU_SUITE_CUSOLVER_CUDA_LINKS SOURCE "${_probe}" LIBRARIES CUDA::cudart CUDA::cusolver)
  if(GPU_SUITE_CUSOLVER_CUDA_LINKS)
    set(_cuda_enabled TRUE)
    set(_cuda_reason "")
  else()
    set(_cuda_reason "CUDA Runtime/cuSOLVER getrf/getrs symbols did not link")
  endif()
endif()
if(_cuda_enabled AND GPU_SUITE_BUILD_EXAMPLES)
  add_executable(solver_gpu examples/solver_gpu.cu)
  target_link_libraries(solver_gpu PRIVATE CUDA::cudart CUDA::cusolver)
  gpu_suite_configure_executable(solver_gpu examples/solver_gpu.cu LIBRARY cusolver IMPLEMENTATION cuda ROLE example BUILD_PROFILE cpu-cuda BACKEND_VARIANT cuda-cusolver)
endif()
if(_cuda_enabled AND GPU_SUITE_BUILD_BENCHMARKS)
  add_executable(solver_gpu_bench benchmarks/solver_gpu_bench.cu)
  target_link_libraries(solver_gpu_bench PRIVATE gpu_suite::common CUDA::cudart CUDA::cusolver)
  gpu_suite_configure_executable(solver_gpu_bench benchmarks/solver_gpu_bench.cu LIBRARY cusolver IMPLEMENTATION cuda ROLE benchmark BUILD_PROFILE cpu-cuda BACKEND_VARIANT cuda-cusolver)
endif()
set(_openacc_enabled FALSE)
set(_openacc_reason "OpenACC targets are disabled")
if(GPU_SUITE_OPENACC_TARGETS_ENABLED AND TARGET OpenACC::OpenACC_CXX AND TARGET CUDA::cudart AND TARGET CUDA::cusolver)
  gpu_suite_check_cxx_link(GPU_SUITE_CUSOLVER_OPENACC_LINKS SOURCE "${_probe}" LIBRARIES OpenACC::OpenACC_CXX CUDA::cudart CUDA::cusolver)
  if(GPU_SUITE_CUSOLVER_OPENACC_LINKS)
    set(_openacc_enabled TRUE)
    set(_openacc_reason "")
  else()
    set(_openacc_reason "nvc++ OpenACC CUDA Runtime/cuSOLVER probe failed")
  endif()
endif()
if(_openacc_enabled AND GPU_SUITE_BUILD_EXAMPLES)
  add_executable(openacc_cusolver examples/openacc_cusolver.cpp)
  target_link_libraries(openacc_cusolver PRIVATE OpenACC::OpenACC_CXX CUDA::cudart CUDA::cusolver)
  gpu_suite_configure_executable(openacc_cusolver examples/openacc_cusolver.cpp LIBRARY cusolver IMPLEMENTATION openacc ROLE example BUILD_PROFILE openacc BACKEND_VARIANT nvhpc-cusolver)
endif()
if(_openacc_enabled AND GPU_SUITE_BUILD_BENCHMARKS)
  add_executable(openacc_cusolver_bench benchmarks/openacc_cusolver_bench.cpp)
  target_link_libraries(openacc_cusolver_bench PRIVATE gpu_suite::common OpenACC::OpenACC_CXX CUDA::cudart CUDA::cusolver)
  gpu_suite_configure_executable(openacc_cusolver_bench benchmarks/openacc_cusolver_bench.cpp LIBRARY cusolver IMPLEMENTATION openacc ROLE benchmark BUILD_PROFILE openacc BACKEND_VARIANT nvhpc-cusolver)
endif()
gpu_suite_record_status("cuSOLVER CPU targets" "${_cpu_enabled}" "${_cpu_reason}")
gpu_suite_record_status("cuSOLVER CUDA targets" "${_cuda_enabled}" "${_cuda_reason}")
gpu_suite_record_status("cuSOLVER OpenACC targets" "${_openacc_enabled}" "${_openacc_reason}")
