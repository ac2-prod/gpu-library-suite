if(GPU_SUITE_BUILD_CPU)
  if(GPU_SUITE_BUILD_EXAMPLES)
    add_executable(rand_cpu examples/rand_cpu.cpp)
    gpu_suite_configure_executable(rand_cpu examples/rand_cpu.cpp
      LIBRARY rocrand IMPLEMENTATION cpu ROLE example BUILD_PROFILE cpu
      BACKEND_VARIANT std-random-serial CPU_BACKENDS cpu-std-random-serial)
  endif()
  if(GPU_SUITE_BUILD_BENCHMARKS)
    add_executable(rand_cpu_bench benchmarks/rand_cpu_bench.cpp)
    target_link_libraries(rand_cpu_bench PRIVATE gpu_suite::common)
    gpu_suite_configure_executable(rand_cpu_bench benchmarks/rand_cpu_bench.cpp
      LIBRARY rocrand IMPLEMENTATION cpu ROLE benchmark BUILD_PROFILE cpu
      BACKEND_VARIANT std-random-serial CPU_BACKENDS cpu-std-random-serial)
  endif()
endif()
foreach(_implementation hip openmp)
  if(_implementation STREQUAL hip)
    set(_enabled "${GPU_SUITE_ROCRAND_HIP_ENABLED}")
    set(_stem rand_gpu)
  else()
    set(_enabled "${GPU_SUITE_ROCRAND_OPENMP_ENABLED}")
    set(_stem openmp_rocrand)
  endif()
  if(_enabled)
    foreach(_role example benchmark)
      if(_role STREQUAL example)
        set(_target "${_stem}")
        set(_source "examples/${_target}.cpp")
        set(_requested "${GPU_SUITE_BUILD_EXAMPLES}")
      else()
        set(_target "${_stem}_bench")
        set(_source "benchmarks/${_target}.cpp")
        set(_requested "${GPU_SUITE_BUILD_BENCHMARKS}")
      endif()
      if(_requested)
        add_executable(${_target} "${_source}")
        if(_implementation STREQUAL hip)
          set_source_files_properties("${_source}" PROPERTIES LANGUAGE HIP)
        endif()
        target_link_libraries(${_target} PRIVATE RocRAND::rocrand RocRAND::hip)
        if(_role STREQUAL benchmark)
          target_link_libraries(${_target} PRIVATE gpu_suite::common)
        endif()
        gpu_suite_configure_executable(${_target} "${_source}"
          LIBRARY rocrand IMPLEMENTATION "${_implementation}" ROLE "${_role}"
          BUILD_PROFILE "${_implementation}" BACKEND_VARIANT "${_implementation}-rocrand")
      endif()
    endforeach()
  endif()
endforeach()
gpu_suite_record_status("rocRAND CPU targets" "${GPU_SUITE_BUILD_CPU}" "dependency-free std::mt19937_64")
gpu_suite_record_status("rocRAND HIP targets" "${GPU_SUITE_ROCRAND_HIP_ENABLED}" "${GPU_SUITE_ROCRAND_HIP_DISABLE_REASON}")
gpu_suite_record_status("rocRAND OpenMP targets" "${GPU_SUITE_ROCRAND_OPENMP_ENABLED}" "${GPU_SUITE_ROCRAND_OPENMP_DISABLE_REASON}")
