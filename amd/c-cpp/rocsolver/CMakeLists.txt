include(GpuSuiteCpuDependencies)
include(GpuSuiteProbe)

set(_cpu_enabled FALSE)
set(_cpu_reason "GPU_SUITE_BUILD_CPU is OFF")
if(GPU_SUITE_BUILD_CPU)
  if(GPU_SUITE_LAPACKE_FOUND)
    set(_cpu_enabled TRUE)
    set(_cpu_reason "")
    string(TOLOWER "${GPU_SUITE_LAPACKE_PROVIDER}" _cpu_variant)
    if(_cpu_variant STREQUAL "generic_lapacke")
      set(_cpu_backend cpu-generic-lapacke)
    elseif(_cpu_variant STREQUAL "openblas")
      set(_cpu_backend cpu-openblas)
    else()
      set(_cpu_backend cpu-onemkl)
    endif()
  else()
    set(_cpu_reason "${GPU_SUITE_LAPACKE_REASON}")
  endif()
endif()
if(_cpu_enabled AND GPU_SUITE_BUILD_EXAMPLES)
  add_executable(solver_cpu examples/solver_cpu.c)
  target_link_libraries(solver_cpu PRIVATE GpuSuite::LAPACKE)
  gpu_suite_configure_executable(solver_cpu examples/solver_cpu.c LIBRARY rocsolver IMPLEMENTATION cpu ROLE example BUILD_PROFILE cpu BACKEND_VARIANT "${_cpu_variant}" CPU_BACKENDS "${_cpu_backend}")
endif()
if(_cpu_enabled AND GPU_SUITE_BUILD_BENCHMARKS)
  add_executable(solver_cpu_bench benchmarks/solver_cpu_bench.c)
  target_link_libraries(solver_cpu_bench PRIVATE gpu_suite::common GpuSuite::LAPACKE)
  target_compile_definitions(solver_cpu_bench PRIVATE "GPU_SUITE_COMPILED_CPU_BACKEND=\"${_cpu_backend}\"")
  gpu_suite_configure_executable(solver_cpu_bench benchmarks/solver_cpu_bench.c LIBRARY rocsolver IMPLEMENTATION cpu ROLE benchmark BUILD_PROFILE cpu BACKEND_VARIANT "${_cpu_variant}" CPU_BACKENDS "${_cpu_backend}")
endif()


foreach(_implementation hip openmp)
  if(_implementation STREQUAL hip)
    set(_enabled "${GPU_SUITE_ROCSOLVER_HIP_ENABLED}")
    set(_stem solver_gpu)
  else()
    set(_enabled "${GPU_SUITE_ROCSOLVER_OPENMP_ENABLED}")
    set(_stem openmp_rocsolver)
  endif()
  if(_enabled)
    foreach(_role example benchmark)
      if(_role STREQUAL example)
        set(_target "${_stem}")
        set(_source "examples/${_target}.cpp")
        set(_requested "${GPU_SUITE_BUILD_EXAMPLES}")
      else()
        set(_target "${_stem}_bench")
        set(_source "benchmarks/${_target}.cpp")
        set(_requested "${GPU_SUITE_BUILD_BENCHMARKS}")
      endif()
      if(_requested)
        add_executable(${_target} "${_source}")
        if(_implementation STREQUAL hip)
          set_source_files_properties("${_source}" PROPERTIES LANGUAGE HIP)
        endif()
        target_link_libraries(${_target} PRIVATE RocSOLVER::rocsolver RocSOLVER::rocblas RocSOLVER::hip)
        if(_role STREQUAL benchmark)
          target_link_libraries(${_target} PRIVATE gpu_suite::common)
        endif()
        gpu_suite_configure_executable(${_target} "${_source}"
          LIBRARY rocsolver IMPLEMENTATION "${_implementation}" ROLE "${_role}"
          BUILD_PROFILE "${_implementation}" BACKEND_VARIANT "${_implementation}-rocsolver")
      endif()
    endforeach()
  endif()
endforeach()
gpu_suite_record_status("rocSOLVER CPU targets" "${_cpu_enabled}" "${_cpu_reason}")
gpu_suite_record_status("rocSOLVER HIP targets" "${GPU_SUITE_ROCSOLVER_HIP_ENABLED}" "${GPU_SUITE_ROCSOLVER_HIP_DISABLE_REASON}")
gpu_suite_record_status("rocSOLVER OpenMP targets" "${GPU_SUITE_ROCSOLVER_OPENMP_ENABLED}" "${GPU_SUITE_ROCSOLVER_OPENMP_DISABLE_REASON}")
