set(_cpu_enabled "${GPU_SUITE_BUILD_CPU}")
set(_cpu_variant stl-serial)
set(_cpu_backend cpu-stl-serial)
set(_cpu_reason "dependency-free STL")
if(GPU_SUITE_BUILD_CPU AND GPU_SUITE_CPU_THRUST_BACKEND STREQUAL "OPENMP")
  find_package(OpenMP COMPONENTS CXX QUIET)
  if(OpenMP_CXX_FOUND)
    set(_cpu_variant openmp)
    set(_cpu_backend cpu-openmp)
    set(_cpu_reason "OpenMP CXX compile-and-link support")
  else()
    set(_cpu_enabled FALSE)
    set(_cpu_reason "OpenMP CXX runtime unavailable, requested backend is not replaced")
  endif()
elseif(GPU_SUITE_BUILD_CPU AND NOT GPU_SUITE_CPU_THRUST_BACKEND STREQUAL "STL")
  message(FATAL_ERROR "GPU_SUITE_CPU_THRUST_BACKEND must be STL or OPENMP")
endif()
if(_cpu_enabled)
  foreach(_role example benchmark)
    if(_role STREQUAL example)
      set(_target reduce_cpu)
      set(_source examples/reduce_cpu.cpp)
      set(_requested "${GPU_SUITE_BUILD_EXAMPLES}")
    else()
      set(_target reduce_cpu_bench)
      set(_source benchmarks/reduce_cpu_bench.cpp)
      set(_requested "${GPU_SUITE_BUILD_BENCHMARKS}")
    endif()
    if(_requested)
      add_executable(${_target} "${_source}")
      if(_role STREQUAL benchmark)
        target_link_libraries(${_target} PRIVATE gpu_suite::common)
        target_compile_definitions(${_target} PRIVATE "GPU_SUITE_COMPILED_CPU_BACKEND=\"${_cpu_backend}\"")
      endif()
      if(_cpu_variant STREQUAL openmp)
        target_link_libraries(${_target} PRIVATE OpenMP::OpenMP_CXX)
        target_compile_definitions(${_target} PRIVATE GPU_SUITE_USE_OPENMP=1)
      endif()
      gpu_suite_configure_executable(${_target} "${_source}"
        LIBRARY rocthrust IMPLEMENTATION cpu ROLE "${_role}" BUILD_PROFILE cpu
        BACKEND_VARIANT "${_cpu_variant}" CPU_BACKENDS "${_cpu_backend}")
    endif()
  endforeach()
endif()
if(GPU_SUITE_ROCTHRUST_OPENMP_ENABLED AND
   (GPU_SUITE_BUILD_EXAMPLES OR GPU_SUITE_BUILD_BENCHMARKS))
  # Explicit second translation unit: HIP instantiates rocThrust; the caller is
  # compiled with OpenMP target flags. Both compile commands enter provenance.
  set_source_files_properties(examples/rocthrust_wrapper.hip PROPERTIES LANGUAGE HIP)
  add_library(rocthrust_openmp_wrapper OBJECT examples/rocthrust_wrapper.hip)
  target_link_libraries(rocthrust_openmp_wrapper PRIVATE RocThrust::rocthrust RocThrust::hip)
  set_target_properties(rocthrust_openmp_wrapper PROPERTIES HIP_STANDARD 17 HIP_STANDARD_REQUIRED YES)
endif()
foreach(_implementation hip openmp)
  if(_implementation STREQUAL hip)
    set(_enabled "${GPU_SUITE_ROCTHRUST_HIP_ENABLED}")
    set(_stem reduce_gpu)
  else()
    set(_enabled "${GPU_SUITE_ROCTHRUST_OPENMP_ENABLED}")
    set(_stem openmp_rocthrust)
  endif()
  if(_enabled)
    foreach(_role example benchmark)
      if(_role STREQUAL example)
        set(_target "${_stem}")
        set(_source "examples/${_target}.cpp")
        set(_requested "${GPU_SUITE_BUILD_EXAMPLES}")
      else()
        set(_target "${_stem}_bench")
        set(_source "benchmarks/${_target}.cpp")
        set(_requested "${GPU_SUITE_BUILD_BENCHMARKS}")
      endif()
      if(_requested)
        add_executable(${_target} "${_source}")
        if(_implementation STREQUAL hip)
          set_source_files_properties("${_source}" PROPERTIES LANGUAGE HIP)
        else()
          target_link_libraries(${_target} PRIVATE rocthrust_openmp_wrapper)
          # HIP object libraries otherwise propagate a higher linker preference.
          # The final driver must support the caller's AMD OpenMP target flags.
          set_target_properties(${_target} PROPERTIES LINKER_LANGUAGE CXX)
        endif()
        target_link_libraries(${_target} PRIVATE RocThrust::rocthrust RocThrust::hip)
        if(_role STREQUAL benchmark)
          target_link_libraries(${_target} PRIVATE gpu_suite::common)
        endif()
        gpu_suite_configure_executable(${_target} "${_source}"
          LIBRARY rocthrust IMPLEMENTATION "${_implementation}" ROLE "${_role}"
          BUILD_PROFILE "${_implementation}" BACKEND_VARIANT "${_implementation}-rocthrust")
      endif()
    endforeach()
  endif()
endforeach()
gpu_suite_record_status("rocThrust CPU targets" "${_cpu_enabled}" "${_cpu_reason}")
gpu_suite_record_status("rocThrust HIP targets" "${GPU_SUITE_ROCTHRUST_HIP_ENABLED}" "${GPU_SUITE_ROCTHRUST_HIP_DISABLE_REASON}")
gpu_suite_record_status("rocThrust OpenMP targets" "${GPU_SUITE_ROCTHRUST_OPENMP_ENABLED}" "${GPU_SUITE_ROCTHRUST_OPENMP_DISABLE_REASON}")
