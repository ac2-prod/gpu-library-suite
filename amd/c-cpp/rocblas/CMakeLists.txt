include(GpuSuiteCpuDependencies)
include(GpuSuiteProbe)

set(_cpu_enabled FALSE)
set(_cpu_reason "GPU_SUITE_BUILD_CPU is OFF")
if(GPU_SUITE_BUILD_CPU)
  if(GPU_SUITE_CBLAS_FOUND)
    set(_cpu_enabled TRUE)
    set(_cpu_reason "")
    string(TOLOWER "${GPU_SUITE_CBLAS_PROVIDER}" _cpu_variant)
    if(_cpu_variant STREQUAL "generic_cblas")
      set(_cpu_backend cpu-generic-cblas)
    elseif(_cpu_variant STREQUAL "openblas")
      set(_cpu_backend cpu-openblas)
    else()
      set(_cpu_backend cpu-onemkl)
    endif()
  else()
    set(_cpu_reason "${GPU_SUITE_CBLAS_REASON}")
  endif()
endif()
if(_cpu_enabled AND GPU_SUITE_BUILD_EXAMPLES)
  add_executable(blas_cpu examples/blas_cpu.c)
  target_link_libraries(blas_cpu PRIVATE GpuSuite::CBLAS)
  gpu_suite_configure_executable(blas_cpu examples/blas_cpu.c
    LIBRARY rocblas IMPLEMENTATION cpu ROLE example BUILD_PROFILE cpu
    BACKEND_VARIANT "${_cpu_variant}" CPU_BACKENDS "${_cpu_backend}")
endif()
if(_cpu_enabled AND GPU_SUITE_BUILD_BENCHMARKS)
  add_executable(blas_cpu_bench benchmarks/blas_cpu_bench.c)
  target_link_libraries(blas_cpu_bench PRIVATE gpu_suite::common GpuSuite::CBLAS)
  target_compile_definitions(blas_cpu_bench PRIVATE
    "GPU_SUITE_COMPILED_CPU_BACKEND=\"${_cpu_backend}\"")
  gpu_suite_configure_executable(blas_cpu_bench benchmarks/blas_cpu_bench.c
    LIBRARY rocblas IMPLEMENTATION cpu ROLE benchmark BUILD_PROFILE cpu
    BACKEND_VARIANT "${_cpu_variant}" CPU_BACKENDS "${_cpu_backend}")
endif()

foreach(_implementation hip openmp)
  string(TOUPPER "${_implementation}" _upper)
  set(_enabled "${GPU_SUITE_ROCBLAS_${_upper}_ENABLED}")
  if(_implementation STREQUAL hip)
    set(_stem blas_gpu)
  else()
    set(_stem openmp_rocblas)
  endif()
  if(_enabled)
    foreach(_role example benchmark)
      if(_role STREQUAL example)
        set(_target "${_stem}")
        set(_source "examples/${_target}.cpp")
        set(_requested "${GPU_SUITE_BUILD_EXAMPLES}")
      else()
        set(_target "${_stem}_bench")
        set(_source "benchmarks/${_target}.cpp")
        set(_requested "${GPU_SUITE_BUILD_BENCHMARKS}")
      endif()
      if(_requested)
        add_executable(${_target} "${_source}")
        if(_implementation STREQUAL hip)
          set_source_files_properties("${_source}" PROPERTIES LANGUAGE HIP)
        endif()
        target_link_libraries(${_target} PRIVATE RocBLAS::rocblas RocBLAS::hip)
        if(_role STREQUAL benchmark)
          target_link_libraries(${_target} PRIVATE gpu_suite::common)
        endif()
        gpu_suite_configure_executable(${_target} "${_source}"
          LIBRARY rocblas IMPLEMENTATION "${_implementation}" ROLE "${_role}"
          BUILD_PROFILE "${_implementation}" BACKEND_VARIANT "${_implementation}-rocblas")
      endif()
    endforeach()
  endif()
  gpu_suite_record_status("RocBLAS ${_implementation} targets" "${_enabled}"
    "${GPU_SUITE_ROCBLAS_${_upper}_DISABLE_REASON}")
endforeach()
gpu_suite_record_status("RocBLAS CPU targets" "${_cpu_enabled}" "${_cpu_reason}")
