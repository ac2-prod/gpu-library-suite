include(GpuSuiteCpuDependencies)
include(GpuSuiteProbe)

set(_cpu_enabled FALSE)
set(_cpu_reason "GPU_SUITE_BUILD_CPU is OFF")
if(GPU_SUITE_BUILD_CPU)
  if(GPU_SUITE_CPU_SPARSE_BACKEND STREQUAL "REFERENCE")
    set(_cpu_enabled TRUE)
    set(_cpu_variant reference-csr)
    set(_cpu_backend cpu-reference-csr)
    set(_cpu_reason "")
  elseif(GPU_SUITE_CPU_SPARSE_BACKEND STREQUAL "AOCLSPARSE")
    if(GPU_SUITE_AOCL_SPARSE_FOUND)
      set(_cpu_enabled TRUE)
      set(_cpu_variant aocl-sparse)
      set(_cpu_backend cpu-aocl-sparse)
      set(_cpu_reason "")
    else()
      set(_cpu_reason "${GPU_SUITE_AOCL_SPARSE_REASON}")
    endif()
  elseif(GPU_SUITE_CPU_SPARSE_BACKEND STREQUAL "ONEMKL")
    if(GPU_SUITE_MKL_SPARSE_FOUND)
      set(_cpu_enabled TRUE)
      set(_cpu_variant onemkl-sparse)
      set(_cpu_backend cpu-onemkl)
      set(_cpu_reason "")
    else()
      set(_cpu_reason "${GPU_SUITE_MKL_SPARSE_REASON}")
    endif()
  else()
    set(_cpu_reason "invalid GPU_SUITE_CPU_SPARSE_BACKEND")
  endif()
endif()
set(_cpu_example_enabled "${_cpu_enabled}")
set(_cpu_example_reason "${_cpu_reason}")
if(_cpu_example_enabled AND GPU_SUITE_BUILD_EXAMPLES)
  add_executable(sparse_cpu examples/sparse_cpu.c)
  if(_cpu_backend STREQUAL "cpu-onemkl")
    target_link_libraries(sparse_cpu PRIVATE GpuSuite::MKLSparse)
    target_compile_definitions(sparse_cpu PRIVATE GPU_SUITE_USE_MKL_SPARSE=1)
  elseif(_cpu_backend STREQUAL "cpu-aocl-sparse")
    target_link_libraries(sparse_cpu PRIVATE GpuSuite::AOCLSparse)
    target_compile_definitions(sparse_cpu PRIVATE GPU_SUITE_USE_AOCL_SPARSE=1)
  endif()
  gpu_suite_configure_executable(sparse_cpu examples/sparse_cpu.c
    LIBRARY rocsparse IMPLEMENTATION cpu ROLE example BUILD_PROFILE cpu
    BACKEND_VARIANT "${_cpu_variant}" CPU_BACKENDS "${_cpu_backend}")
endif()
if(_cpu_enabled AND GPU_SUITE_BUILD_BENCHMARKS)
  add_executable(sparse_cpu_bench benchmarks/sparse_cpu_bench.c)
  target_link_libraries(sparse_cpu_bench PRIVATE gpu_suite::common)
  if(_cpu_backend STREQUAL "cpu-onemkl")
    target_link_libraries(sparse_cpu_bench PRIVATE GpuSuite::MKLSparse)
    target_compile_definitions(sparse_cpu_bench PRIVATE GPU_SUITE_USE_MKL_SPARSE=1)
  elseif(_cpu_backend STREQUAL "cpu-aocl-sparse")
    target_link_libraries(sparse_cpu_bench PRIVATE GpuSuite::AOCLSparse)
    target_compile_definitions(sparse_cpu_bench PRIVATE GPU_SUITE_USE_AOCL_SPARSE=1)
  endif()
  target_compile_definitions(sparse_cpu_bench PRIVATE
    "GPU_SUITE_COMPILED_CPU_BACKEND=\"${_cpu_backend}\"")
  gpu_suite_configure_executable(sparse_cpu_bench benchmarks/sparse_cpu_bench.c
    LIBRARY rocsparse IMPLEMENTATION cpu ROLE benchmark BUILD_PROFILE cpu
    BACKEND_VARIANT "${_cpu_variant}" CPU_BACKENDS "${_cpu_backend}")
endif()

foreach(_implementation hip openmp)
  string(TOUPPER "${_implementation}" _upper)
  set(_enabled "${GPU_SUITE_ROCSPARSE_${_upper}_ENABLED}")
  if(_implementation STREQUAL hip)
    set(_stem sparse_gpu)
  else()
    set(_stem openmp_rocsparse)
  endif()
  if(_enabled)
    foreach(_role example benchmark)
      if(_role STREQUAL example)
        set(_target "${_stem}")
        set(_source "examples/${_target}.cpp")
        set(_requested "${GPU_SUITE_BUILD_EXAMPLES}")
      else()
        set(_target "${_stem}_bench")
        set(_source "benchmarks/${_target}.cpp")
        set(_requested "${GPU_SUITE_BUILD_BENCHMARKS}")
      endif()
      if(_requested)
        add_executable(${_target} "${_source}")
        if(_implementation STREQUAL hip)
          set_source_files_properties("${_source}" PROPERTIES LANGUAGE HIP)
        endif()
        target_link_libraries(${_target} PRIVATE RocSPARSE::rocsparse RocSPARSE::hip)
        if(_role STREQUAL benchmark)
          target_link_libraries(${_target} PRIVATE gpu_suite::common)
        endif()
        gpu_suite_configure_executable(${_target} "${_source}"
          LIBRARY rocsparse IMPLEMENTATION "${_implementation}" ROLE "${_role}"
          BUILD_PROFILE "${_implementation}" BACKEND_VARIANT "${_implementation}-rocsparse")
      endif()
    endforeach()
  endif()
  gpu_suite_record_status("RocSPARSE ${_implementation} targets" "${_enabled}"
    "${GPU_SUITE_ROCSPARSE_${_upper}_DISABLE_REASON}")
endforeach()
gpu_suite_record_status("RocSPARSE CPU targets" "${_cpu_enabled}" "${_cpu_reason}")
