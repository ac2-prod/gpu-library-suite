# Dependency resolution happens before build metadata is captured.
set(_rocfft_cpu_reason "GPU_SUITE_BUILD_CPU is OFF")
set(_rocfft_cpu_enabled FALSE)
if(GPU_SUITE_BUILD_CPU)
  if(FFTW3f_FOUND)
    set(_rocfft_cpu_enabled TRUE)
    set(_rocfft_cpu_reason "")
  elseif(CMAKE_DISABLE_FIND_PACKAGE_FFTW3f)
    set(_rocfft_cpu_reason "FFTW3f discovery explicitly disabled by CMAKE_DISABLE_FIND_PACKAGE_FFTW3f")
  elseif(FFTW3f_DISABLE_REASON)
    set(_rocfft_cpu_reason "${FFTW3f_DISABLE_REASON}")
  else()
    set(_rocfft_cpu_reason "FFTW3f dependency was not found")
  endif()
endif()
if(_rocfft_cpu_enabled AND GPU_SUITE_BUILD_EXAMPLES)
  add_executable(fft_cpu examples/fft_cpu.c)
  target_link_libraries(fft_cpu PRIVATE FFTW3f::fftw3f)
  gpu_suite_configure_executable(fft_cpu examples/fft_cpu.c
    LIBRARY rocfft IMPLEMENTATION cpu ROLE example
    BUILD_PROFILE cpu BACKEND_VARIANT fftw3f-serial CPU_BACKENDS cpu-fftw-serial)
endif()
if(_rocfft_cpu_enabled AND GPU_SUITE_BUILD_BENCHMARKS)
  add_executable(fft_cpu_bench benchmarks/fft_cpu_bench.c)
  target_link_libraries(fft_cpu_bench PRIVATE gpu_suite::common FFTW3f::fftw3f)
  set(_rocfft_cpu_variant fftw3f-serial)
  set(_rocfft_cpu_backends cpu-fftw-serial)
  if(FFTW3f_THREADS_FOUND)
    target_compile_definitions(fft_cpu_bench PRIVATE GPU_SUITE_HAVE_FFTW_THREADS=1)
    target_link_libraries(fft_cpu_bench PRIVATE FFTW3f::threads)
    set(_rocfft_cpu_variant fftw3f-serial+threaded)
    list(APPEND _rocfft_cpu_backends cpu-fftw-threaded)
  endif()
  gpu_suite_configure_executable(fft_cpu_bench benchmarks/fft_cpu_bench.c
    LIBRARY rocfft IMPLEMENTATION cpu ROLE benchmark
    BUILD_PROFILE cpu BACKEND_VARIANT "${_rocfft_cpu_variant}"
    CPU_BACKENDS ${_rocfft_cpu_backends})
endif()

foreach(_implementation hip openmp)
  if(_implementation STREQUAL hip)
    set(_enabled "${GPU_SUITE_HIP_TARGETS_ENABLED}")
    set(_stem fft_gpu)
    set(_variant hip-rocfft)
  else()
    set(_enabled "${GPU_SUITE_OPENMP_TARGETS_ENABLED}")
    set(_stem openmp_rocfft)
    set(_variant openmp-rocfft)
  endif()
  if(_enabled)
    foreach(_role example benchmark)
      if(_role STREQUAL example)
        set(_target "${_stem}")
        set(_source "examples/${_target}.cpp")
        set(_requested "${GPU_SUITE_BUILD_EXAMPLES}")
      else()
        set(_target "${_stem}_bench")
        set(_source "benchmarks/${_target}.cpp")
        set(_requested "${GPU_SUITE_BUILD_BENCHMARKS}")
      endif()
      if(_requested)
        add_executable(${_target} "${_source}")
        if(_implementation STREQUAL hip)
          set_source_files_properties("${_source}" PROPERTIES LANGUAGE HIP)
        endif()
        target_link_libraries(${_target} PRIVATE RocFFT::rocfft RocFFT::hip)
        if(_role STREQUAL benchmark)
          target_link_libraries(${_target} PRIVATE gpu_suite::common)
        endif()
        gpu_suite_configure_executable(${_target} "${_source}"
          LIBRARY rocfft IMPLEMENTATION "${_implementation}" ROLE "${_role}"
          BUILD_PROFILE "${_implementation}" BACKEND_VARIANT "${_variant}")
      endif()
    endforeach()
  endif()
endforeach()

gpu_suite_record_status("rocFFT CPU targets" "${_rocfft_cpu_enabled}" "${_rocfft_cpu_reason}")
gpu_suite_record_status("rocFFT threaded CPU series" "${FFTW3f_THREADS_FOUND}" "${FFTW3f_THREADS_DISABLE_REASON}")
gpu_suite_record_status("rocFFT HIP targets" "${GPU_SUITE_HIP_TARGETS_ENABLED}" "${GPU_SUITE_HIP_DISABLE_REASON}")
gpu_suite_record_status("rocFFT OpenMP targets" "${GPU_SUITE_OPENMP_TARGETS_ENABLED}" "${GPU_SUITE_OPENMP_DISABLE_REASON}")
