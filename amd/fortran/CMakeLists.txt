# Separate build tree, canonical source stems, native Fortran workloads.
set(_common "${CMAKE_CURRENT_SOURCE_DIR}/common")
add_library(gpu_suite_amd_fortran_support STATIC "${_common}/benchmark_support.f90")
set(_support_modules "${CMAKE_CURRENT_BINARY_DIR}/modules/support")
file(MAKE_DIRECTORY "${_support_modules}")
set_target_properties(gpu_suite_amd_fortran_support PROPERTIES
  Fortran_MODULE_DIRECTORY "${_support_modules}")
target_include_directories(gpu_suite_amd_fortran_support PUBLIC "${_support_modules}")
target_compile_options(gpu_suite_amd_fortran_support PRIVATE ${GPU_SUITE_FORTRAN_SAFE_OPTIONS})

set(_libraries rocfft rocblas rocsparse rocsolver rocrand rocthrust)
set(_packages RocFFT RocBLAS RocSPARSE RocSOLVER RocRAND RocThrust)
set(_stems fft blas sparse solver rand reduce)
foreach(_index RANGE 0 5)
  list(GET _libraries ${_index} _library)
  if(NOT _library IN_LIST GPU_SUITE_AMD_LIBRARIES)
    continue()
  endif()
  list(GET _packages ${_index} _package)
  list(GET _stems ${_index} _stem)
  foreach(_impl RANGE 0 2)
    set(_enabled FALSE)
    set(_reason "not requested")
    set(_links "")
    set(_includes "")
    set(_options "")
    set(_helpers "")
    set(_wrapper "")
    set(_backend "")
    set(_definitions "GPU_SUITE_IMPL=${_impl}")
    if(_index EQUAL 3)
      list(APPEND _helpers "${CMAKE_SOURCE_DIR}/common/fortran/make_dense_system.f90")
    endif()
    if(_impl EQUAL 0)
      set(_implementation cpu)
      set(_name "${_stem}_cpu")
      if(GPU_SUITE_BUILD_CPU)
        set(_enabled TRUE)
        set(_reason "")
        if(_index EQUAL 0)
          set(_backend cpu-fftw-serial)
          if(FFTW3f_FOUND AND GPU_SUITE_FORTRAN_FFTW_INCLUDE)
            set(_links FFTW3f::fftw3f)
            set(_includes "${GPU_SUITE_FORTRAN_FFTW_INCLUDE}")
            list(APPEND _definitions GPU_SUITE_FORTRAN_FFTW=1)
          else()
            set(_enabled FALSE)
            set(_reason "FFTW3f library and fftw3.f03 are required")
          endif()
        elseif(_index EQUAL 1 OR _index EQUAL 3)
          if(_index EQUAL 1)
            set(_provider CBLAS)
            set(_choice "${GPU_SUITE_CPU_BLAS_BACKEND}")
            set(_generic cpu-generic-cblas)
          else()
            set(_provider LAPACKE)
            set(_choice "${GPU_SUITE_CPU_LAPACK_BACKEND}")
            set(_generic cpu-generic-lapacke)
          endif()
          if(_choice STREQUAL ONEMKL)
            set(_backend cpu-onemkl)
          elseif(_choice STREQUAL OPENBLAS)
            set(_backend cpu-openblas)
          else()
            set(_backend "${_generic}")
          endif()
          if(TARGET GpuSuite::${_provider})
            set(_links GpuSuite::${_provider})
          else()
            set(_enabled FALSE)
            set(_reason "${GPU_SUITE_${_provider}_REASON}")
          endif()
        elseif(_index EQUAL 2)
          set(_backend cpu-aocl-sparse)
          if(TARGET GpuSuite::AOCLSparse)
            set(_links GpuSuite::AOCLSparse)
            list(APPEND _definitions GPU_SUITE_FORTRAN_AOCL_SPARSE=1)
            list(APPEND _helpers "${CMAKE_CURRENT_SOURCE_DIR}/rocsparse/aocl_sparse_c.f90")
          else()
            set(_enabled FALSE)
            set(_reason "${GPU_SUITE_AOCL_SPARSE_REASON}")
          endif()
        elseif(_index EQUAL 4)
          set(_backend cpu-fortran-random-serial)
        else()
          set(_backend cpu-fortran-sum-serial)
        endif()
      endif()
    else()
      if(_impl EQUAL 1)
        set(_implementation hip)
        set(_name "${_stem}_gpu")
        set(_enabled "${GPU_SUITE_HIP_TARGETS_ENABLED}")
        set(_reason "${GPU_SUITE_HIP_DISABLE_REASON}")
      else()
        set(_implementation openmp)
        set(_name "openmp_${_library}")
        set(_enabled "${GPU_SUITE_OPENMP_TARGETS_ENABLED}")
        set(_reason "${GPU_SUITE_OPENMP_DISABLE_REASON}")
        set(_options "${GPU_SUITE_OPENMP_COMPILE_OPTIONS}")
      endif()
      if(_enabled)
        if(NOT ${_package}_FOUND)
          set(_enabled FALSE)
          set(_reason "${_package} headers/library not found")
        else()
          set(_links GpuSuite::hipfort "${_package}::${_library}" "${_package}::hip")
          set(_includes "${GPU_SUITE_HIPFORT_MODULE_DIR}")
          if(_index EQUAL 3)
            list(APPEND _links RocSOLVER::rocblas)
          elseif(_index EQUAL 2)
            list(APPEND _helpers "${CMAKE_CURRENT_SOURCE_DIR}/rocsparse/rocsparse_v2_c.f90")
          elseif(_index EQUAL 5)
            if(GPU_SUITE_FORTRAN_HIP_WRAPPER_ENABLED)
              set(_wrapper "${CMAKE_CURRENT_SOURCE_DIR}/rocthrust/examples/rocthrust_wrapper.hip")
            else()
              set(_enabled FALSE)
              set(_reason "rocThrust needs CMake 3.29+, native HIP Clang and explicit CMAKE_HIP_ARCHITECTURES")
            endif()
          endif()
        endif()
      endif()
    endif()
    if(_index EQUAL 2)
      list(APPEND _helpers "${CMAKE_CURRENT_SOURCE_DIR}/rocsparse/make_poisson2d_csr.f90")
    endif()
    set(_source "${CMAKE_CURRENT_SOURCE_DIR}/${_library}/examples/${_name}.f90")
    if(_enabled)
      set(_probe_linker "")
      if(_wrapper)
        set(_probe_linker LINKER_LANGUAGE Fortran)
      endif()
      gpu_suite_fortran_probe(_amd_probe_${_name} SOURCES ${_helpers} "${_source}" ${_wrapper}
        LIBRARIES ${_links} INCLUDES ${_includes} OPTIONS ${_options} ${_probe_linker})
      if(NOT _amd_probe_${_name})
        set(_enabled FALSE)
        set(_reason "Fortran API/module compile/link failed; see fortran-probes/_amd_probe_${_name}.log")
      endif()
    endif()
    if(_enabled)
      foreach(_role example benchmark)
        if(_role STREQUAL example)
          if(NOT GPU_SUITE_BUILD_EXAMPLES)
            continue()
          endif()
          set(_target "${_name}")
          set(_target_source "${_source}")
          set(_sources "${_source}" ${_helpers} ${_wrapper})
        else()
          if(NOT GPU_SUITE_BUILD_BENCHMARKS)
            continue()
          endif()
          set(_target "${_name}_bench")
          set(_target_source "${CMAKE_CURRENT_SOURCE_DIR}/${_library}/benchmarks/${_target}.f90")
          set(_sources "${_target_source}" ${_helpers} ${_wrapper}
            "${CMAKE_CURRENT_SOURCE_DIR}/${_library}/benchmarks/${_stem}_workload.F90"
            "${_common}/benchmark_bridge.cpp")
          if(NOT _impl EQUAL 0)
            list(APPEND _sources "${_common}/hip_device_bridge.cpp")
          endif()
        endif()
        add_executable(${_target} ${_sources})
        set(_modules "${CMAKE_CURRENT_BINARY_DIR}/modules/${_target}")
        file(MAKE_DIRECTORY "${_modules}")
        set_target_properties(${_target} PROPERTIES Fortran_MODULE_DIRECTORY "${_modules}"
          LINKER_LANGUAGE Fortran Fortran_PREPROCESS ON
          RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/${_library}")
        target_include_directories(${_target} BEFORE PRIVATE "${_modules}" ${_includes})
        target_compile_definitions(${_target} PRIVATE ${_definitions}
          "GPU_SUITE_FORTRAN_CPU_BACKEND=\"${_backend}\""
          "$<$<COMPILE_LANGUAGE:CXX>:_POSIX_C_SOURCE=200809L>")
        foreach(_option IN LISTS GPU_SUITE_FORTRAN_SAFE_OPTIONS)
          target_compile_options(${_target} PRIVATE "$<$<COMPILE_LANGUAGE:Fortran>:${_option}>")
        endforeach()
        target_link_options(${_target} PRIVATE ${GPU_SUITE_FORTRAN_SAFE_OPTIONS})
        target_link_libraries(${_target} PRIVATE ${_links})
        set(_backends "${_backend}")
        if(_role STREQUAL benchmark)
          target_link_libraries(${_target} PRIVATE gpu_suite_amd_fortran_support gpu_suite::common)
          if(_impl EQUAL 0 AND _index EQUAL 0 AND FFTW3f_THREADS_FOUND)
            target_link_libraries(${_target} PRIVATE FFTW3f::threads)
            target_compile_definitions(${_target} PRIVATE GPU_SUITE_HAVE_FFTW_THREADS=1)
            list(APPEND _backends cpu-fftw-threaded)
          endif()
        endif()
        set(_variant "fortran-${_implementation}-${_library}")
        if(_backend STREQUAL "cpu-aocl-sparse")
          set(_variant "aocl-sparse")
        endif()
        gpu_suite_configure_executable(${_target} "${_target_source}"
          LIBRARY "${_library}" IMPLEMENTATION "${_implementation}" ROLE "${_role}"
          BUILD_PROFILE "${_implementation}" BACKEND_VARIANT "${_variant}"
          CPU_BACKENDS ${_backends})
      endforeach()
    endif()
    gpu_suite_record_status("AMD Fortran ${_library} ${_implementation}" "${_enabled}" "${_reason}")
  endforeach()
endforeach()
